find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Network Concurrent)
get_target_property(qt_core_type Qt6::Core TYPE)
if(qt_core_type STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "Manager requires dynamically linked Qt. Supply shared Qt via CMAKE_PREFIX_PATH.")
endif()

qt_standard_project_setup(REQUIRES 6.5)
if(Qt6_VERSION VERSION_GREATER_EQUAL 6.8)
    qt_policy(SET QTP0004 NEW)
endif()
qt_add_executable(AetherWall.Manager main.cpp ManagerController.cpp ManagerController.h DebugLog.cpp DebugLog.h CoreProcess.cpp CoreProcess.h CatalogClient.cpp CatalogClient.h CatalogNetwork.cpp CatalogNetwork.h)
qt_add_qml_module(AetherWall.Manager URI AetherWall.Manager VERSION 1.0
    RESOURCE_PREFIX /qt/qml QML_FILES qml/Main.qml qml/Discover.qml qml/AwButton.qml qml/AwTextField.qml qml/AwComboBox.qml qml/AwCheckBox.qml qml/AwLabel.qml RESOURCES assets/aetherwall.svg assets/aetherwall.png)
target_include_directories(AetherWall.Manager PRIVATE "${CMAKE_SOURCE_DIR}/src")
target_link_libraries(AetherWall.Manager PRIVATE aetherwall_control Qt6::Core Qt6::Gui Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Concurrent)
file(GLOB_RECURSE catalog_mock_files CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/server/mock/*")
qt_add_resources(AetherWall.Manager "catalog_mock" PREFIX "/catalog" BASE "${CMAKE_SOURCE_DIR}/server/mock" FILES ${catalog_mock_files})
if(WIN32)
    target_sources(AetherWall.Manager PRIVATE ../core/resources.rc)
    target_include_directories(AetherWall.Manager PRIVATE ../core)
    target_link_libraries(AetherWall.Manager PRIVATE aetherwall_pipe shell32)
    set_target_properties(AetherWall.Manager PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
if(MSVC)
    target_compile_options(AetherWall.Manager PRIVATE /W4 /permissive- /utf-8 /EHsc)
else()
    target_compile_options(AetherWall.Manager PRIVATE -Wall -Wextra -Wpedantic)
endif()

# The network/package suite uses in-process storage and works without a desktop.
# Keep the Unix socket harness separate from the production Windows Named Pipe.
if(BUILD_TESTING AND WIN32)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(aetherwall_catalog_tests ../../tests/manager/CatalogTests.cpp CatalogClient.cpp CatalogClient.h CatalogNetwork.cpp CatalogNetwork.h)
    target_include_directories(aetherwall_catalog_tests PRIVATE . "${CMAKE_SOURCE_DIR}/src" "${CMAKE_SOURCE_DIR}/tests/manager")
    target_compile_definitions(aetherwall_catalog_tests PRIVATE "AETHERWALL_BUNDLED_DIR=L\"${CMAKE_SOURCE_DIR}/scenes/bundled\"")
    target_compile_options(aetherwall_catalog_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
    target_link_libraries(aetherwall_catalog_tests PRIVATE aetherwall_control Qt6::Core Qt6::Gui Qt6::Network Qt6::Concurrent Qt6::Test)
    file(GLOB_RECURSE windows_catalog_files CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/server/mock/*")
    qt_add_resources(aetherwall_catalog_tests "catalog_mock_windows" PREFIX "/catalog" BASE "${CMAKE_SOURCE_DIR}/server/mock" FILES ${windows_catalog_files})
    add_test(NAME Catalog.DownloadAndValidation COMMAND aetherwall_catalog_tests
        -o -,txt -o "${CMAKE_CURRENT_BINARY_DIR}/catalog-results.txt,txt")
    # The test runs before windeployqt. Resolve DLLs from the exact linked Qt kit
    # for both local CTest and CI without relying on the caller's PATH.
    set_tests_properties(Catalog.DownloadAndValidation PROPERTIES
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
endif()
